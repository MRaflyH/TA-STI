# Modelling environment -- QNN and classical-NN baseline.
#
# SEPARATE FROM dataset-pipeline/requirements.txt ON PURPOSE. That environment
# does network I/O (cdsapi, requests, xarray, netcdf4); this one does not touch
# the network at all. Keeping them apart means installing qiskit can never
# disturb a download that is mid-flight, and the pipeline can be rebuilt on a
# machine where the quantum stack does not install cleanly.
#
#     python3 -m venv .venv
#     source .venv/bin/activate          # Windows: .venv\Scripts\activate
#     pip install -r requirements.txt
#
# Verified working together on Python 3.12. Python 3.10+ is required
# (qiskit-machine-learning 0.9 dropped 3.9).

# --------------------------------------------------------------------------
# Quantum stack -- PINNED EXACTLY, and the pins matter
# --------------------------------------------------------------------------
# qiskit-machine-learning 0.8.x pins qiskit<2.0; 0.9.x requires qiskit>=2.0.
# An unpinned `pip install qiskit qiskit-machine-learning` can therefore
# resolve to either side of that boundary and fail in ways that look like a
# code bug rather than a resolution problem. The API also changed across it:
# 2.x circuit-library constructors are snake_case functions (efficient_su2),
# the primitives are V2, and V1 primitives are gone. gfd_model targets the
# 2.x / 0.9.x side.
qiskit==2.5.2
qiskit-machine-learning==0.9.1
qiskit-aer==0.17.2

# --------------------------------------------------------------------------
# Classical ML and data
# --------------------------------------------------------------------------
# scikit-learn supplies the MLPRegressor baseline and the whole shared
# preprocessing chain (imputer, Yeo-Johnson, min-max, PCA) that both models
# consume, so its version is part of what makes a result reproducible (NF-02).
scikit-learn==1.8.0
numpy==2.4.4
scipy==1.17.1
pandas==3.0.2

# Reading the processed tables the pipeline writes.
pyarrow==25.0.1

# --------------------------------------------------------------------------
# Figures
# --------------------------------------------------------------------------
