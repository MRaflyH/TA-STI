\section{Petir dan Ground Flash Density}

\begin{itemize}
    \item Proses pembentukan muatan pada awan konvektif dan mekanisme sambaran
          ke tanah.
          % CITE: buku teks fisika petir
    \item Klasifikasi sambaran: \textit{cloud-to-ground} (CG) dan
          \textit{intra-cloud} (IC); hanya CG yang relevan bagi proteksi
          infrastruktur di permukaan.
          % CITE: klasifikasi CG vs IC dan proporsi tipikalnya

\section{Jaringan Deteksi Petir}

\begin{itemize}
    \item Prinsip lokalisasi sambaran: \textit{time-of-arrival} dan
          \textit{magnetic direction finding}, serta peran jumlah sensor.
          % CITE: prinsip kerja jaringan deteksi petir
    \item \textit{Detection efficiency} (DE) dan ketergantungannya pada jarak
          dari sensor; DE yang menurun terbaca sebagai aktivitas petir yang
          menurun.
          % CITE: studi detection efficiency jaringan deteksi petir
    \item PLN Puslitbang LDS sebagai sumber domain tropis; medan yang tersedia
          mencakup \texttt{Discrimination}, arus puncak, dan \texttt{Multi.}
          (D-22).
    \item NASA MERLIN melalui KSC Weather Archive sebagai sumber domain
          subtropis; \textbf{tidak ada kolom diskriminasi} dan tidak ada kolom
          multiplisitas (D-23).
          % CITE: dokumentasi KSC Weather Archive, produk MerlinCloudToGround
          %   (inilah pustaka yang akan menyelesaikan pertanyaan terbuka 10)
    \item Gradien DE terukur pada domain subtropis: korelasi Spearman
          $-0,955$ antara jumlah \textit{flash} dan jarak dari Cape Canaveral,
          melintasi empat orde besaran (D-27).
\end{itemize}

\section{Pemodelan Target yang Didominasi Nol}

\begin{itemize}
    \item Sifat target \textit{zero-inflated} dan mengapa
          R\textsuperscript{2} menjadi nyaris tak bermakna di atasnya:
          memprediksi nol di mana-mana sudah menjelaskan sebagian besar
          variansnya (D-20).
          % CITE: pemodelan data count zero-inflated
    \item \textbf{Hurdle model} sebagai dua tahap: okurensi (biner), lalu
          intensitas bersyarat pada okurensi (D-13).
          % CITE: hurdle model / two-part model
    \item Transformasi \texttt{log1p} pada target cacah, dan
          \textit{retransformation bias}: nilai harap di ruang log yang
          di-\texttt{expm1} adalah besaran mirip median, dan menurut
          ketaksamaan Jensen berada di bawah rerata bersyarat cacahnya
          (D-03).
          % CITE: ketaksamaan Jensen dalam konteks retransformasi
    \item \textbf{Duan smearing estimator} sebagai koreksi multiplikatif
          satu parameter atas bias tersebut.
          % CITE: Duan (smearing estimator)
    \item Pentingnya \textit{baseline} trivial: setiap tabel metrik memuat skor
          model \enquote{prediksi nol/laju klimatologis} dalam satuan yang sama
          (D-07, D-12).
\end{itemize}

\section{Komputasi Kuantum dan Quantum Machine Learning}

\begin{itemize}
    \item Konsep dasar: qubit, superposisi, keterbelitan, gerbang berparameter,
          dan nilai harap operator Pauli yang hidup pada rentang $[-1, 1]$.
          % CITE: buku teks komputasi kuantum
    \item \textit{Feature map} dan penyandian sudut (\textit{angle encoding});
          alasan rentang masukan harus terbatas, dan mengapa sudut melewati
          $\pi$ ber-\textit{alias} kembali ke sudut lain (D-35).
          % CITE: angle encoding / data encoding pada QML
    \item \textit{Variational quantum circuit} dan \textit{ansatz}
          \texttt{real\_amplitudes}; hubungan jumlah bobot dengan
          $(\text{reps}+1) \times n$ (D-33).
          % CITE: variational quantum algorithms / VQC
          % CITE: ansatz real_amplitudes
    \item \textbf{Barren plateau}: gradien yang meluruh eksponensial terhadap
          jumlah qubit, dan mitigasinya lewat \textbf{fungsi biaya lokal} ---
          dasar pemilihan \texttt{observable = local\_mean} alih-alih
          \texttt{global\_z} (D-33).
          % CITE: barren plateau
          % CITE: local cost function sebagai mitigasi barren plateau
    \item Metode diferensiasi: \textit{parameter-shift} lawan \textit{adjoint};
          keduanya menghasilkan gradien, dengan biaya komputasi yang sangat
          berbeda (D-28).
          % CITE: parameter-shift rule
          % CITE: adjoint differentiation
    \item Klaim efisiensi sampel pada QML, yang pada TA ini \textbf{diuji dan
          tidak didukung} (D-14) --- lihat Bab~\ref{chap:evaluasi}.
          % CITE: klaim keunggulan QML pada data terbatas
    \item Perkakas: Qiskit dan PennyLane, serta perbedaan peran keduanya pada TA
          ini (D-28, D-39).
          % CITE: Qiskit
          % CITE: PennyLane
\end{itemize}

          bulanan (D-21), dan dua domain iklim alih-alih satu.
    \item Perbedaan terhadap literatur QML terapan: perbandingan dilakukan pada

\end{itemize}
